import React, { useCallback, useMemo, useRef, useState } from "react";
import {
  GoogleMap,
  InfoWindow,
  LoadScript,
  Marker,
  Polyline,
} from "@react-google-maps/api";
import { applyTalkingTreesVoice } from "../utils/talkingTreesVoice";

// Woodland POIs
const POIS = [
  { name: "Meeting Point", lat: 44.62664, lng: -63.92356, description: "Gather here before starting the walk through the woodland." },
  { name: "Starting Point", lat: 44.62664, lng: -63.92336, description: "Trail entrance where the main route begins." },
  { name: "Exercising Machine", lat: 44.62639, lng: -63.92353, description: "Outdoor exercise station along the forest route." },
  { name: "The Well", lat: 44.62625, lng: -63.92347, description: "Historic well — an old water source from early homestead life." },
  { name: "The Farm House", lat: 44.62611, lng: -63.923, description: "Former homestead site from the area's early settlers." },
  { name: "Setting Area", lat: 44.62589, lng: -63.92261, description: "Quiet seating area surrounded by trees." },
  { name: "First Phone", lat: 44.62583, lng: -63.92266, description: "First listening phone for the Talking Trees experience." },
  { name: "Yellow Birch", lat: 44.6255, lng: -63.92181, description: "Coastal Yellow Birch — important for biodiversity." },
  { name: "Fung architecture", lat: 44.62514, lng: -63.92122, description: "Woodland fungi along bark patterns." },
  { name: "Second Phone", lat: 44.62503, lng: -63.92119, description: "Second listening phone for more forest stories." },
  { name: "Labyrinth starting point", lat: 44.62417, lng: -63.91958, description: "Entrance to the woodland labyrinth path." },
  { name: "Middle of the Labyrinth", lat: 44.62392, lng: -63.91969, description: "Center of the woodland labyrinth." },
  { name: "Lake View", lat: 44.62028, lng: -63.91672, description: "Scenic overlook of the lake." },
];

const containerStyle = { width: "100%", height: "600px" };

// Natural map look
const mapOptions = {
  mapTypeId: "terrain",
  streetViewControl: false,
  mapTypeControl: false,
  fullscreenControl: true,
  zoomControl: true,
};

const SiteMap = () => {
  const mapRef = useRef(null);
  const [selectedPoi, setSelectedPoi] = useState(null);
  const [talkingTreesStarted, setTalkingTreesStarted] = useState(false);
  const [userLocation, setUserLocation] = useState(null);

  const mapCenter = useMemo(() => {
    const sum = POIS.reduce(
      (acc, poi) => {
        acc.lat += poi.lat;
        acc.lng += poi.lng;
        return acc;
      },
      { lat: 0, lng: 0 }
    );
    return { lat: sum.lat / POIS.length, lng: sum.lng / POIS.length };
  }, []);

  const speakSoftly = useCallback(
    (text) => {
      if (!text || !talkingTreesStarted) return;
      window.speechSynthesis.cancel();
      const utter = applyTalkingTreesVoice(new SpeechSynthesisUtterance(text));
      window.speechSynthesis.speak(utter);
    },
    [talkingTreesStarted]
  );

  const handleMarkerClick = useCallback(
    (poi) => {
      setSelectedPoi(poi);
      speakSoftly(poi.description || poi.name);
    },
    [speakSoftly]
  );

  const handlePlayAudio = () => {
    if (!selectedPoi) return;
    speakSoftly(selectedPoi.description || selectedPoi.name);
  };

  const handleLocateUser = () => {
    if (!navigator.geolocation) {
      alert("Geolocation not supported.");
      return;
    }
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        const coords = { lat: pos.coords.latitude, lng: pos.coords.longitude };
        setUserLocation(coords);

        if (mapRef.current) {
          mapRef.current.panTo(coords);
          mapRef.current.setZoom(17);
        }
      },
      (err) => {
        alert("Location unavailable or permission denied.");
        console.warn(err);
      },
      { enableHighAccuracy: true, timeout: 10000 }
    );
  };

  const handleGetDirectionsClick = (poi) => {
    const url = `https://www.google.com/maps/dir/?api=1&destination=${poi.lat},${poi.lng}&travelmode=walking`;
    window.open(url, "_blank");
  };

  return (
    <div className="p-8 bg-lime-800 dark:bg-darkBlue text-gray-900 dark:text-gray-100 min-h-screen flex flex-col items-center justify-center">
      <h1 className="text-3xl font-sans mb-4 text-white dark:text-gray-300">
        Woodland Conservation Area
      </h1>

      <div className="flex flex-col md:flex-row gap-3 items-center mb-4">
        <button
          onClick={() => setTalkingTreesStarted(true)}
          className="px-4 py-2 bg-yellow-500 hover:bg-yellow-600 text-black rounded"
        >
          Start Talking Trees
        </button>

        <button
          onClick={handleLocateUser}
          className="px-4 py-2 bg-orange-900 hover:bg-orange-800 text-white rounded"
        >
          YOU ARE HERE
        </button>
      </div>

      <div className="w-full max-w-5xl">
        <LoadScript googleMapsApiKey={process.env.REACT_APP_GOOGLE_MAPS_API_KEY}>
          <GoogleMap
            onLoad={(map) => (mapRef.current = map)}
            mapContainerStyle={containerStyle}
            center={mapCenter}
            zoom={16}
            options={mapOptions}
          >
            {/* Woodland Hiking Trail — Earth-Tone Dashed */}
            <Polyline
              path={POIS.map((poi) => ({ lat: poi.lat, lng: poi.lng }))}
              options={{
                strokeOpacity: 0, // hide solid line
                icons: [
                  {
                    icon: {
                      path: "M 0,-1 0,1",
                      scale: 4,
                      strokeOpacity: 1,
                      strokeColor: "#8B5A2B", // natural brown trail color
                    },
                    offset: "0",
                    repeat: "20px",
                  },
                ],
              }}
            />

            {/* POI Markers */}
            {POIS.map((poi) => (
              <Marker
                key={poi.name}
                position={{ lat: poi.lat, lng: poi.lng }}
                onClick={() => handleMarkerClick(poi)}
              />
            ))}

            {/* User Location Marker */}
            {userLocation && (
              <Marker
                position={userLocation}
                icon={{
                  url: "http://maps.google.com/mapfiles/ms/icons/blue-dot.png",
                }}
              />
            )}

            {/* Info Popup */}
            {selectedPoi && (
              <InfoWindow
                position={{ lat: selectedPoi.lat, lng: selectedPoi.lng }}
                onCloseClick={() => setSelectedPoi(null)}
              >
                <div className="text-black">
                  <div className="font-semibold mb-1">{selectedPoi.name}</div>
                  <p className="mb-2 text-sm max-w-xs">{selectedPoi.description}</p>

                  <div className="flex gap-2">
                    <button
                      onClick={handlePlayAudio}
                      className="px-3 py-1 bg-green-700 text-white rounded"
                    >
                      Play Audio
                    </button>

                    <button
                      onClick={() => handleGetDirectionsClick(selectedPoi)}
                      className="px-3 py-1 bg-blue-600 text-white rounded"
                    >
                      Get Directions
                    </button>
                  </div>
                </div>
              </InfoWindow>
            )}
          </GoogleMap>
        </LoadScript>
      </div>

      <div className="mt-8 text-center text-white dark:text-gray-300">
        <h2 className="text-2xl font-bold mb-2">Instructions</h2>
        <p>Click Start Talking Trees to enable speech narration.</p>
        <p className="mt-2">Tap markers to hear descriptions or get walking directions.</p>
      </div>
    </div>
  );
};

export default SiteMap;
